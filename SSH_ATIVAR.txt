# 1. Cria um arquivo de 2GB para ser usado como memória extra
sudo fallocate -l 2G /swapfile
# 2. Ajusta as permissões de segurança
sudo chmod 600 /swapfile
# 3. Formata o arquivo como Swap
sudo mkswap /swapfile
# 4. Ativa a memória agora
sudo swapon /swapfile
# 5. Faz com que essa memória continue ativa mesmo se reiniciar o servidor
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab
# 6. Verifica se funcionou (deve aparecer uma linha 'Swap')
free -h
